describe("Android Tailnet profile route", () => {
  beforeEach(() => {
    vi.clearAllMocks();
    mocks.consumeLocalRateLimit.mockReturnValue({ allowed: true, retryAfterSeconds: 0 });
    mocks.getTrustedClientRateLimitKey.mockReturnValue(null);
    mocks.generateSubscriptionYaml.mockResolvedValue({
      yaml: "mode: rule\nproxies: []\nproxy-groups: []\nrules: [\"MATCH,DIRECT\"]\n",
      name: "Test", subscriptionInfo: {}, cacheExpirySeconds: 3600,
      autoUpdateIntervalSeconds: 43200, isAdmin: true,
    });
  });

  it("returns the original bytes when no profile is requested", async () => {
    const response = await GET(new Request("https://local.test/config.yaml"), {
      params: Promise.resolve({ id: "secret-token" }),
    });
    expect(await response.text()).toBe("mode: rule\nproxies: []\nproxy-groups: []\nrules: [\"MATCH,DIRECT\"]\n");
  });

  it("keeps upstream update headers for the mobile subscription", async () => {
    const response = await GET(new Request("https://local.test/config.yaml?profile=android-tailscale"), {
      params: Promise.resolve({ id: "secret-token" }),
    });
    expect(response.status).toBe(200);
    expect(await response.text()).toContain("type: tailscale");
    expect(response.headers.get("profile-update-interval")).toBe("12");
    expect(response.headers.get("cache-control")).toBe("no-store");
  });

  it("rejects unknown profiles and invalid subscription tokens", async () => {
    let response = await GET(new Request("https://local.test/config.yaml?profile=typo"), {
      params: Promise.resolve({ id: "secret-token" }),
    });
    expect(response.status).toBe(400);
    mocks.generateSubscriptionYaml.mockResolvedValueOnce(null);
    response = await GET(new Request("https://local.test/config.yaml?profile=android-tailscale"), {
      params: Promise.resolve({ id: "invalid-token" }),
    });
    expect(response.status).toBe(404);
  });
});
